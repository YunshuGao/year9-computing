THE ONE FILE  ·  compliment.html
<html> <head> <!-- CSS here --><style> body{text-align:center;font-family:Arial} h1{color:#3f3399} button{background:#3f3399;color:#fff} table,td,th{border:1px solid #888;border-collapse:collapse;padding:4px 12px} </style> </head>
<body>
  <h1 id="msg">Welcome!</h1>

  <!-- a FORM: type your name -->
  <form> Your name: <input id="name"> </form>

  <button onclick="greet()">Compliment me</button>
  <button onclick="showAll()">Show all</button>
  <button onclick="pick()">Pick one</button>

  <!-- a LIST: the loop fills this -->
  <ul id="list"></ul>

  <!-- a TABLE: the menu -->
  <table>
    <tr><th>No.</th><th>Compliment</th></tr>
    <tr><td>1</td><td>You're a legend!</td></tr>
    <tr><td>2</td><td>Brilliant!</td></tr>
    <tr><td>3</td><td>Future developer!</td></tr>
  </table>
  <!-- the JavaScript will go here -->  <script>
    let lines = ["You're a legend!", "Brilliant!", "Future developer!"];
    function greet() {
      let name = document.getElementById("name").value;
      document.getElementById("msg").innerHTML = "Hi " + name + ", " + lines[0];
    }
    <!-- a LOOP will go here -->    // LOOP — fill the LIST with every compliment
    function showAll() {
      let all = "";
      for (let i = 0; i < lines.length; i = i + 1) {
        all = all + "<li>" + lines[i] + "</li>";
      }
      document.getElementById("list").innerHTML = all;
    }
    <!-- an IF / ELSE will go here -->    // IF / ELSE — YOU pick from the table
    function pick() {
      let choice = prompt("Pick 1, 2 or 3");
      if (choice === "1") { ...innerHTML = lines[0]; }
      else if (choice === "2") { ...innerHTML = lines[1]; }
      else { ...innerHTML = lines[2]; }
    }
  </script>
</body> </html>
