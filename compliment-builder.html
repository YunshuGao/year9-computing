THE ONE FILE  ·  compliment.html
<!DOCTYPE html>
<html> <head>
  <!-- the CSS will go here -->  <style>
    body { text-align: center; font-family: Arial; }
    h1 { color: #3f3399; }   button { background: #3f3399; color: #fff; }
  </style>
</head>
<body>
  <h1 id="msg">Click a button!</h1>
  <button onclick="compliment()">Compliment me</button>
  <button onclick="showAll()">Show all</button>
  <button onclick="lucky()">Feeling lucky?</button>
  <p id="list"></p>
  <!-- the JavaScript will go here -->  <script>
    let lines = ["You're a legend!", "Brilliant!", "Future developer!"];
    function compliment() {
      let pick = lines[Math.floor(Math.random() * lines.length)];
      document.getElementById("msg").innerHTML = pick;
    }
    <!-- a LOOP will go here -->    // LOOP — show EVERY compliment
    function showAll() {
      let all = "";
      for (let i = 0; i < lines.length; i = i + 1) {
        all = all + lines[i] + "<br>";
      }
      document.getElementById("list").innerHTML = all;
    }
    <!-- an IF / ELSE will go here -->    // IF / ELSE — make a decision
    function lucky() {
      let n = Math.floor(Math.random() * 10) + 1;
      if (n >= 5) {
        document.getElementById("msg").innerHTML = n + " — you are a STAR!";
      } else {
        document.getElementById("msg").innerHTML = n + " — keep going!";
      }
    }
  </script>
</body> </html>
